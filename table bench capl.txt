void gotosleep()
{ 
    if(CounterB==2)
    {
     SleepCommand();
     //DAT_E_VCU_422.SYNCHRO_DC_GLOBAL=0;
     Msg_211h.VCU_SYNCHRO_GPC=0;
     SetRcdlowlevel();
    }
    
    if(CounterB==480)
     { 
      PowerOFF();
     }
    
    if(CounterB==482)
    {
      testcaseNum=0xff;
      CounterB=0;
      executing=0;
    }
}

void gotowait()   
{
 
  if(CounterB==10)
  {
  //initial condition// 
   Msg_432h.REVEIL_PRINCIPAL=1;
  /* No main wake up request
  00 : Invalid RCD
  01 : No main wake up request
  10 : Main wake up request
  11 : Not valid*/
   Msg_432h.PRECOND_ELEC_WAKEUP=0;//no wakeup
   Msg_432h.STOP_DELAYED_HMI_WAKEUP=0;
   //Msg_432h.eBCM_INFO_REQUEST_PARTIAL_WAKEUP=0;
   Msg_432h.EBCM_INFO_REQUEST_PARTIAL_WAKEUP=0;
   Msg_432h.BCM_LONG_PARK_WAKEUP=0;
  
  
   Msg_421h.HV_BATT_RECHARGE_WAKEUP=0;//0:No Wakeup 1:Wake up Request
   Msg_421h.PI_STATE_INFO_WAKEUP=0;
   Msg_421h.POST_OPERATION_WAKEUP=0;
 
    /* 
000: No Command
001: DISCHARGE
010: CHARGE
011: PRE-HEAT
100: Reserve
101: Ready
110: HVBAT Precondition
111: Invalid
   */  
   Msg_211h.VCU_STATE_COMMAND=0;
  
 /* 
0: No emergency shutdown request
1: Emergency shutdown request
*/
   Msg_241h.VCU_EMERGENCY_SHUTDOWN=0;//0: No emergency shutdown request 1: Emergency shutdown request
   Msg_241h.VITESSE_VEHICULE_ROUES=0;//speed=0
    
   //Msg_351h.CRASH_AIRBAG=5;
    
  
   }
  
   if(CounterB==12)
    { 
      PowerON();
    }
    
    if(CounterB==14)
    {
      SetRcdhighlevel();
    }
  
  if(CounterB==20)
  {
  testcaseNum=0xff;
  CounterB=0;
  executing=0;
  }
  
}

void MODbusCommandSet(byte relayN,byte ONset)
{
  word crcSum;
  byte m;
  word k;
  
  Tframe[0]=0x01;
  Tframe[1]=0x0f;
  Tframe[2]=0x00;
  Tframe[3]=0x00;
  if(ONset==1)
  {
  if(relayN<8)
  {
  RelayS1=RelayS1|(0x01<<relayN);
  Tframe[5]=RelayS1;
  }
  
   if((relayN>=8)&&(relayN<10))
  {
  RelayS2=RelayS2|(0x01<<(relayN-8));
  Tframe[4]=RelayS2;    
  }
  }
  else
  {
    if(relayN<8)
  {
  RelayS1=RelayS1&(~(0x01<<relayN));
  Tframe[5]=RelayS1;
  }
  
   if((relayN>=8)&&(relayN<10))
  {
  RelayS2=RelayS2&(~(0x01<<(relayN-8)));
  Tframe[4]=RelayS2;    
  }
  
  }
 
  crcSum=MODBUScrc16(Tframe,6);
  Tframe[6]=crcSum>>8;
  Tframe[7]=crcSum&0xff;
  rs232Send(portN,Tframe,8);
  
   for(m=0;m<255;m++)
       for(k=0;k<100;k++)
        ;
   
  write("send out data:%x,%x", Tframe[6],Tframe[7]);
  
}

word MODBUScrc16(byte puchMsg[], byte usDataLen) 
{ 
word  CRCSUM;
word CRCSUMh;
byte i,k,shiftBit,CRCSUMl;
word DATAtemp;
 CRCSUM=0xFFFF;
for(i=0;i<usDataLen;i++)  
{ 
  CRCSUMh=CRCSUM&0xff00;
  CRCSUMl=CRCSUM&0xff;
  CRCSUMl=CRCSUMl^puchMsg[i];
  CRCSUM=CRCSUMh|CRCSUMl;
  for(k=0;k<8;k++)
  {
  shiftBit=CRCSUM&0x01;
  CRCSUM=(CRCSUM>>0x01)&0x7FFF;
  if(shiftBit==0x01) 
    CRCSUM=CRCSUM^0xA001;
  }

}   
 return CRCSUM;

}

void CheckSumCalculation(message *can_msg,byte index,byte Counter,byte CHKINI)
{
  
  byte m;
  byte len;
  WORD DataTmp=0;
  byte ChkSum=0;
  byte constantD[8];
  byte DataItem[16];
  
   for(m=0;m<8;m++)
    constantD[m]=can_msg.byte(m);
  
   len=can_msg.dlc;
  
   constantD[index]=constantD[index]&0x00;
   constantD[index]=(constantD[index]|Counter);
  
   for(m=0;m<len;m++)
  {
    can_msg.byte(m)=constantD[m];
  }
  
  
   for(m=0;m<len;m++)
  {
    DataItem[2*m]=constantD[m]&0xf;
    DataItem[2*m+1]=(constantD[m]&0xf0)>>4;  
  }
  
  DataTmp=0;
  
  for(m=0;m<2*len;m++)
    DataTmp=DataTmp+DataItem[m];

  DataTmp=DataTmp&0x0f;
  
  ChkSum=0x0f-(byte)DataTmp;
   
  can_msg.byte(index)=can_msg.byte(index)|(ChkSum<<4);
